\section{Math and Figures}

\begin{frame}{An equation}
  The least-squares estimate is
  \begin{equation*}
    \yhat = \X\,(\X\tp\X)^{-1}\X\tp\y,
    \qquad \x \in \real^{p},
  \end{equation*}
  and the residual variance is estimated by
  \begin{equation*}
    \hat\sigma^{2} = \frac{1}{n-p}\lVert \y - \yhat \rVert^{2}.
  \end{equation*}
\end{frame}
